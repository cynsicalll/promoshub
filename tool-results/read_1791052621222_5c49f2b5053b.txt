   816→  lang: Lang,
   817→  key: string,
   818→  vars?: Record<string, string | number>
   819→): string {
   820→  const dict = dictionaries[lang] || dictionaries.en;
   821→  let str = dict[key] ?? dictionaries.en[key] ?? key;
   822→  if (vars) {
   823→    for (const [k, v] of Object.entries(vars)) {
   824→      str = str.replace(new RegExp(`\\{${k}\\}`, "g"), String(v));
   825→    }
   826→  }
   827→  return str;
