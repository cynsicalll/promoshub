   107→   107→      })

   114→   114→

   116→   116→  React.useEffect(() => {
   117→   117→    if (!promo) return;

   119→   119→
   120→   120→    async function fetchApprovals() {
   121→   121→      if (!promo) return;

   123→   123→        // Fetch approval status for all payments in this promo
   124→   124→        const results: Record<string, { status: string; respondedAt?: string }> = {};

   126→   126→          const res = await fetch(`/api/payment-approvals?paymentId=${p.id}`);

   131→   131→            const latest = data[0];
   132→   132→            results[p.id] = { status: latest.status, respondedAt: latest.respondedAt };
   133→   133→          }
   134→   134→        }
   135→   135→        if (!cancelled) setApprovalStatuses(results);
   136→   136→      } catch { /* ignore */ }

   139→   139→    fetchApprovals();

   172→   172→    if (field === "paidOut" && value) {
   173→   173→      patch.payoutDate = toISODate(formatDateInput(new Date()));
   174→   174→    }
   175→   175→    if (field === "paidOut" && !value) patch.payoutDate = null;

   177→   177→      await updatePayment.mutateAsync({ id, data: patch as never });

   180→   180→      toast.error(t("detail.updateFailedToast"));
   181→   181→    }

   200→   200→        description: url,
   201→   201→      });
   202→   202→    } catch {
   203→   203→      toast.error(t("detail.shareEnableFailedToast"));
   204→   204→    }
   205→   205→  }
   206→   206→
   207→   207→  async function handleDisableShare() {
   208→   208→    try {
   209→   209→      await disableShare.mutateAsync(promo!.id);
   210→   210→      toast.success(t("detail.shareDisabledToast"));
   211→   211→    } catch {
   212→   212→      toast.error(t("detail.shareDisableFailedToast"));
   213→   213→    }
   214→   214→  }

   216→   216→  async function copyShareLink() {

   220→   220→      await navigator.clipboard.writeText(url);
   221→   221→      toast.success(t("detail.linkCopiedToast"), { description: url });

   223→   223→      toast.error(t("detail.copyFailedToast"));
   224→   224→    }
   225→   225→  }
   226→   226→

   236→   236→            className="mt-0.5"

   259→   259→          {promo.shareEnabled && promo.shareCode ? (

   291→   291→            variant="outline"
   292→   292→            size="sm"
   293→   293→            onClick={() => setFormOpen(true)}
   294→   294→            className="gap-2"

   296→   296→            <Pencil className="h-4 w-4" /> {t("action.edit")}
   297→   297→          </Button>
   298→   298→          {/* Visible to: Viewer (dad) or Collaborator — not both */}
   299→   299→          <div className="flex items-center gap-1.5">
   300→   300→            <span className="text-xs text-muted-foreground whitespace-nowrap">Visible to:</span>
   301→   301→            <Select
   302→   302→              value={(promo as Record<string, unknown>).visibleTo as string || "viewer"}
   303→   303→              onValueChange={async (value) => {

   305→   305→                  const res = await fetch(`/api/promos/${promo.id}/viewer-visible`, {

   315→   315→                  toast.error("Failed to update visibility");
   316→   316→                }
   317→   317→              }}
   318→   318→            >
   319→   319→              <SelectTrigger className="h-8 w-[140px] text-xs">

   338→   338→                const total = promoTotal(promo);

   343→   343→                    type: "promo",
   344→   344→                    data: {
   345→   345→                      name: promo.name,
   346→   346→                      amount: total,
   347→   347→                      customer: promo.customer?.name || "",
   348→   348→                      date: formatDate(promo.date),
   349→   349→                    },
   350→   350→                  }),
   351→   351→                });
   352→   352→                if (!res.ok) throw new Error("Failed");
   353→   353→                const result = await res.json();
   354→   354→                if (result.sent > 0) {
   355→   355→                  toast.success(`Notification sent to viewer: "${result.title}"`);

   358→   358→                }
   359→   359→              } catch {
   360→   360→                toast.error("Failed to send notification");
   361→   361→              }

   364→   364→            <Bell className="h-4 w-4" /> Notify

   372→   372→            <Trash2 className="h-4 w-4" /> {t("action.delete")}

   374→   374→        </div>

   377→   377→      {/* Share banner when enabled */}

   381→   381→            <Share2 className="h-4 w-4" />
   382→   382→          </div>
   383→   383→          <div className="min-w-0 flex-1">
   384→   384→            <p className="text-sm font-medium">{t("detail.shareActive")}</p>
   385→   385→            <p className="text-xs text-muted-foreground truncate">
   386→   386→              {t("detail.shareLink")} {typeof window !== "undefined" ? window.location.origin : ""}/?share={promo.shareCode}
   387→   387→            </p>
   388→   388→          </div>
   389→   389→          <div className="flex gap-2">
   390→   390→            <Button
   391→   391→              variant="outline"
   392→   392→              size="sm"
   393→   393→              onClick={copyShareLink}
   394→   394→              className="gap-2"
   395→   395→            >
   396→   396→              <Copy className="h-3.5 w-3.5" /> {t("action.copy")}
   397→   397→            </Button>
   398→   398→            <a
   399→   399→              href={`/?share=${promo.shareCode}`}
   400→   400→              target="_blank"
   401→   401→              rel="noopener noreferrer"
   402→   402→            >
   403→   403→              <Button variant="outline" size="sm" className="gap-2">
   404→   404→                <ExternalLink className="h-3.5 w-3.5" /> {t("detail.openShare")}
   405→   405→              </Button>
   406→   406→            </a>
   407→   407→          </div>
   408→   408→        </div>
   409→   409→      )}
   410→   410→
   411→   411→      {/* KPIs */}
   412→   412→      <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   413→   413→        <StatCard
   414→   414→          label={t("detail.totalAmount")}

   417→   417→        />
   418→   418→        <StatCard
   419→   419→          label={t("detail.alreadyReceived")}
   420→   420→          value={formatCurrency(received)}
   421→   421→          icon={CheckCircle2}
   422→   422→          tone="emerald"
   423→   423→        />
   424→   424→        <StatCard
   425→   425→          label={t("detail.stillOpen")}
   426→   426→          value={formatCurrency(outstanding)}
   427→   427→          icon={Clock}
   428→   428→          tone={outstanding > 0 ? "amber" : "default"}
   429→   429→        />
   430→   430→        <StatCard
   431→   431→          label={t("detail.soundsRemaining")}
   432→   432→          value={`${soundsRemaining} / ${soundsTotal}`}
   433→   433→          hint={t("detail.soundsUsedHint", { count: soundsUsed })}
   434→   434→          icon={Music2}
   435→   435→          tone="teal"
   436→   436→        />
   437→   437→      </div>
   438→   438→
   439→   439→      <div className="grid gap-5 lg:grid-cols-3">
   440→   440→        {/* Left: Sounds + Customer */}
   441→   441→        <div className="space-y-5">
   442→   442→          <Card className="p-5">
   443→   443→            <h3 className="font-semibold flex items-center gap-2 mb-3">
   444→   444→              <Music2 className="h-4 w-4 text-muted-foreground" /> {t("detail.sounds")}
   445→   445→            </h3>
   446→   446→            {promo.sounds.length === 0 ? (
   447→   447→              <p className="text-sm text-muted-foreground py-2">
   448→   448→                {t("detail.noSounds")}

   450→   450→            ) : (
   451→   451→              <div className="space-y-3">
   452→   452→                {promo.sounds.map((ps) => {
   453→   453→                  const remaining = ps.quantityTotal - ps.quantityUsed;
   454→   454→                  const over = ps.quantityUsed > ps.quantityTotal;
   455→   455→                  const pct =
   456→   456→                    ps.quantityTotal > 0

   458→   458→                          100,
   459→   459→                          (ps.quantityUsed / ps.quantityTotal) * 100

   461→   461→                      : 0;
   462→   462→                  return (
   463→   463→                    <div key={ps.id} className="space-y-1.5">
   464→   464→                      <div className="flex items-center justify-between gap-2">

   472→   472→                        >
   473→   473→                          {ps.quantityUsed}/{ps.quantityTotal}
