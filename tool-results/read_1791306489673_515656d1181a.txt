    43→    43→// Format large numbers (e.g. 1200 -> 1.2K, 1500000 -> 1.5M)
    44→    44→function formatCount(n: number | null | undefined): string {
    45→    45→  if (n == null) return "—";
    46→    46→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
    47→    47→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
    48→    48→  return String(n);
    49→    49→}
